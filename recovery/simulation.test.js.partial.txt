import test from 'node:test';
import assert from 'node:assert/strict';
import { brackets, selectBracket, selectOutcome, continuationProbability, bracketForOop, defaultVoucherPolicy, simulateYear, generatePatient } from './simulation.js';

// Legacy behavior tests hold entry in January; calendar-entry tests are separate.
const generateDynamicPatient = (sequence, policy, random, subfillRandom = random) => generateCalendarPatient(sequence, policy, random, subfillRandom, () => 0);
const withoutVoucher = { eligibleCohortIds: [] };
const cohort = id => brackets.find(bracket => bracket.id === id);
const draws = (...values) => () => values.length ? values.shift() : 0.999999;

// Hold monthly costs constant when isolating the pre-existing voucher and
// behavior rules; separate tests below exercise the actual dynamic schedule.
const simulateYear = (bracket, policy = defaultVoucherPolicy, random = Math.random) => {
  const schedule = createSubfillSchedule(bracket, () => 0.999999).map(fill => ({ ...fill, initialOopCents: bracket.oopCents, preVoucherBracket: bracket }));
  return simulateDynamicYear(bracket, policy, random, schedule);
};
const generatePatient = (sequence, policy = defaultVoucherPolicy, random = Math.random) => generateDynamicPatient(sequence, policy, random, () => 0.999999);

test('all 1,000 probability tickets match the supplied OOP distribution exactly', () => {
  assert.equal(brackets.reduce((total, bracket) => total + bracket.weight, 0), 1000);
  const counts = Object.fromEntries(brackets.map(bracket => [bracket.id, 0]));
  for (let ticket = 0; ticket < 1000; ticket++) counts[selectBracket((ticket + 0.5) / 1000).id]++;
  for (const bracket of brackets) assert.equal(counts[bracket.id], bracket.weight, bracket.id);
  assert.equal(selectBracket(0).id, 'B');
  assert.equal(selectBracket(0.999999999).id, 'R');
});

test('every bracket retains the supplied initial abandonment rate', () => {
  const expectedRates = [978, 1672, 2342, 2361, 2506, 2535, 2917, 4852, 6382, 7619, 7983, 7983, 8062, 8062, 8101, 8809, 8809];
  brackets.forEach((bracket, index) => {
    let abandoned = 0;
    assert.equal(bracket.abandonmentBasisPoints, expectedRates[index]);
    for (let ticket = 0; ticket < 10000; ticket++) {
      if (selectOutcome(bracket, (ticket + 0.5) / 10000) === 'Abandoned') abandoned++;
    }
    assert.equal(abandoned, expectedRates[index], bracket.id);
  });
});

test('conditional persistence reproduces every cumulative cohort curve', () => {
  for (const bracket of brackets) {
    assert.equal(bracket.retention.length, 12);
    assert.equal(bracket.retention[0], 100);
    assert.equal(bracket.retention[11], 0);
    let surviving = 1;
    for (let fill = 2; fill <= 12; fill++) {
      const chance = continuationProbability(bracket, fill);
      assert.ok(chance >= 0 && chance <= 1);
      surviving *= chance;
      assert.ok(Math.abs(surviving - bracket.retention[fill - 1] / 100) < 1e-12, `${bracket.id} fill ${fill}`);
    }
  }
});

test('default vouchers cover all original cohorts and select $0 behavior', () => {
  for (const bracket of brackets) {
    const fills = simulateYear(bracket, defaultVoucherPolicy, () => 0.5);
    assert.equal(fills[0].filled, true);
    assert.equal(fills[1].filled, true); // 50% draw is below A's 64% continuation.
    for (const fill of fills.filter(fill => fill.attempted)) {
      assert.equal(fill.voucherOffered, true);
      assert.equal(fill.effectiveOopCents, 0);
      assert.equal(fill.behaviorBracket.id, 'A');
    }
    assert.equal(fills[0].voucherCostCents, bracket.oopCents);
    assert.ok(Math.abs(fills[0].decisionProbability - 0.9022) < 1e-12);
  }
});

test('voucher precedes abandonment: the same patient fills at $0 and abandons at original OOP', () => {
  const assisted = generatePatient(1, defaultVoucherPolicy, draws(0.2, 0.2));
  const unassisted = generatePatient(2, withoutVoucher, draws(0.2, 0.2));
  assert.equal(assisted.oopCents, 4005);
  assert.equal(assisted.bracket.id, 'G');
  assert.equal(assisted.fills[0].behaviorBracket.id, 'A');
  assert.equal(assisted.outcome, 'Filled');
  assert.equal(unassisted.outcome, 'Abandoned');
  assert.equal(unassisted.fillCount, 0);
  assert.equal(unassisted.fills.length, 12);
});

test('ineligible cohorts retain original OOP and behavior', () => {
  for (const policy of [withoutVoucher, { eligibleCohortIds: ['R'] }]) {
    const fills = simulateYear(cohort('G'), policy, () => 0.5);
    assert.equal(fills[0].effectiveOopCents, 4005);
    assert.equal(fills[0].behaviorBracket.id, 'G');
    assert.equal(fills[0].voucherOffered, false);
    assert.equal(fills[1].filled, false); // G: 44%, not A: 64%.
    assert.equal(fills[0].voucherCostCents, 0);
  }
});

test('later persistence is conditional, without a second abandonment draw', () => {
  const fills = simulateYear(cohort('G'), defaultVoucherPolicy, draws(0.99, 0.63, 0.43));
  assert.equal(fills[1].filled, true);
  assert.equal(fills[2].filled, false); // A: 27/64 = 42.1875%, not 27% or 64%.
  assert.equal(fills[2].decisionProbability, 27 / 64);
});

test('post-voucher OOP maps to the correct cost cohort', () => {
  for (const [cents, expected] of [[0,'A'],[1,'B'],[500,'B'],[501,'C'],[2500,'F'],[2501,'G'],[130000,'Q'],[130001,'R']]) assert.equal(bracketForOop(cents).id, expected);
});

test('no receipt or voucher spending after stopping, including an abandoned first offer', () => {
  let count = 0;
  const fills = simulateYear(cohort('G'), defaultVoucherPolicy, () => { count++; return 0; });
  assert.equal(count, 1);
  assert.equal(fills[0].status, 'Abandoned');
  assert.equal(fills[0].voucherOffered, true);
  assert.equal(fills[0].voucherCostCents, 0);
  assert.ok(fills.slice(1).every(fill => !fill.attempted && !fill.filled && fill.voucherCostCents === 0 && fill.baseDraw === undefined));
});

test('patient rules and records are immutable snapshots', () => {
  const policy = { ...defaultVoucherPolicy, eligibleCohortIds: ['G'] };
  const patient = generatePatient(1, policy, () => 0.5);
  policy.eligibleCohortIds.length = 0;
  assert.deepEqual(patient.voucherPolicy.eligibleCohortIds, ['G']);
  assert.ok(Object.isFrozen(patient.fills[0]));
  assert.ok(Object.isFrozen(patient.fills));
  assert.equal(patient.fillCount, patient.fills.filter(fill => fill.filled).length);
});

test('invalid policy, OOP, and random draws are rejected', () => {
  for (const policy of [{ receiptBasisPoints: 5000, eligibleCohortIds: [] }, { targetOopCents: 2000, eligibleCohortIds: [] }, { behaviorBasis: 'original', eligibleCohortIds: [] }, { eligibleCohortIds: ['N'] }]) assert.throws(() => simulateYear(cohort('G'), policy), RangeError);
  for (const value of [-1, 1, NaN, Infinity]) {
    assert.throws(() => selectBracket(value), RangeError);
    assert.throws(() => simulateYear(cohort('G'), defaultVoucherPolicy, () => value), RangeError);
  }
  assert.throws(() => bracketForOop(-1), RangeError);
});
